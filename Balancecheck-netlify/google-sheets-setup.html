Google Sheets Setup

One Google Sheet runs your whole back office: orders land in it automatically, you set your wallets and email in it, and the Admin page lets you mark orders valid / invalid / delivered with a tap. Takes about 10 minutes, no coding.

Step 1 — Create the Sheet and paste the script

  1. Go to sheets.google.com → create a blank spreadsheet, name it Balancecheck.
  2. Menu Extensions → Apps Script. Delete everything in the editor, paste the script below, press Save (💾).
var ORDERS = "Orders", SETTINGS = "Settings";
var HEAD = ["Date","Order","Type","Brand","Region","Value","Code","PIN","Email","Status","Note"];

function sheet(name) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var s = ss.getSheetByName(name);
  if (!s) {
    s = ss.insertSheet(name);
    if (name === ORDERS) { s.appendRow(HEAD); s.getRange("B:I").setNumberFormat("@"); s.setFrozenRows(1); }
    if (name === SETTINGS) {
      [["adminKey", "change-me-" + Math.random().toString(36).slice(2, 8)],
       ["notifyEmail", ""], ["trc20", ""], ["erc20", ""], ["btc", ""],
       ["btcUsd", ""], ["whatsapp", ""]].forEach(function (r) { s.appendRow(r); });
    }
  }
  return s;
}
function settings() {
  var rows = sheet(SETTINGS).getDataRange().getValues(), o = {};
  rows.forEach(function (r) { if (r[0]) o[String(r[0])] = r[1] == null ? "" : String(r[1]); });
  return o;
}
function ok(x) { return ContentService.createTextOutput(JSON.stringify(x)).setMimeType(ContentService.MimeType.JSON); }
function auth(k) { var a = settings().adminKey; return !!a && k === a; }
function find(id) {
  var v = sheet(ORDERS).getDataRange().getValues();
  for (var i = 1; i < v.length; i++) if (String(v[i][1]) === String(id)) return { row: v[i], n: i + 1 };
  return null;
}

function doGet(e) {
  var p = e.parameter || {}, a = p.action || "ping";
  if (a === "settings") { var s = settings(); delete s.adminKey; return ok({ ok: true, settings: s }); }
  if (a === "status") { var f = find(p.id); return ok(f ? { ok: true, status: f.row[9], note: f.row[10], type: f.row[2] } : { ok: false }); }
  if (a === "list") {
    if (!auth(p.key)) return ok({ ok: false, error: "bad key" });
    var v = sheet(ORDERS).getDataRange().getValues(), out = [];
    for (var i = 1; i < v.length; i++) {
      var r = v[i];
      out.push({ date: r[0], order: r[1], type: r[2], brand: r[3], region: r[4], value: r[5],
                 code: r[6], pin: r[7], email: r[8], status: r[9], note: r[10] });
    }
    return ok({ ok: true, orders: out.reverse() });
  }
  sheet(ORDERS); sheet(SETTINGS);
  return ok({ ok: true, status: "ready" });
}

function doPost(e) {
  var d = JSON.parse(e.postData.contents || "{}"), a = d.action || "order";
  if (a === "order") {
    var s = sheet(ORDERS);
    s.appendRow([new Date(), d.order || "", d.type || "", d.brand || "", d.region || "",
                 d.value || "", d.code || "", d.pin || "", d.email || "", "Pending", ""]);
    return ok({ ok: true });
  }
  if (!auth(d.key)) return ok({ ok: false, error: "bad key" });
  if (a === "status") {
    var f = find(d.order); if (!f) return ok({ ok: false, error: "not found" });
    var o = sheet(ORDERS);
    o.getRange(f.n, 10).setValue(d.status || "Pending");
    o.getRange(f.n, 11).setValue(d.note || "");
    return ok({ ok: true });
  }
  if (a === "settings") {
    var sh = sheet(SETTINGS), v = sh.getDataRange().getValues();
    Object.keys(d.settings || {}).forEach(function (k) {
      if (k === "adminKey") return;
      var hit = false;
      for (var i = 0; i < v.length; i++) if (v[i][0] === k) { sh.getRange(i + 1, 2).setValue(d.settings[k]); hit = true; }
      if (!hit) sh.appendRow([k, d.settings[k]]);
    });
    return ok({ ok: true });
  }
  return ok({ ok: false, error: "unknown action" });
}

Step 2 — Deploy it as a Web app

  1. Click Deploy → New deployment.
  2. Click the ⚙️ gear next to "Select type" → Web app.
  3. Set Execute as: Me and Who has access: Anyone. Click Deploy.
  4. Click Authorize access → choose your Google account → Advanced → Go to (unsafe) → Allow. (It's your own script; Google shows this for every unpublished script.)
  5. Copy the Web app URL. It looks like https://script.google.com/macros/s/AKfyc…/exec
  6. Open that URL once in a new tab. You should see {"ok":true,"status":"ready"} — this also creates the Orders and Settings tabs in your Sheet.

Step 3 — Put the URL in your site

  1. On GitHub open Balancecheck-netlify/site-config.js, click the ✏️ pencil, paste the URL between the quotes:
    sheetsUrl: "https://script.google.com/macros/s/AKfyc…/exec"
  2. Click Commit changes. Netlify redeploys automatically in about a minute.

Important: whenever you edit the script later, you must go to Deploy → Manage deployments → ✏️ → Version: New version → Deploy. The URL stays the same.

Step 4 — Open your Admin page

  1. In your Sheet, open the Settings tab and copy the value next to adminKey. Change it to something private if you like.
  2. Go to yoursite.com/admin.html, paste the key, press Connect.
  3. Fill in your wallet addresses, the email that should receive card photos and codes, and the BTC price. Press Save settings. Buyers now see your addresses at checkout.

How the day-to-day works

CHECK A customer submits a card → the code, PIN, front & back photos go to your email, and a row appears on the Admin page. Check the balance, then tap ✓ Balance valid or ✕ Card invalid. Optionally type a note first (e.g. "Balance: $48.20").

BUY A purchase appears with the buyer's email. Send the gift card to that email, then tap ✓ Delivered.

POP-UPS Every tap shows a confirmation pop-up on your side. The buyer's order page checks your Sheet every 15 seconds and shows a pop-up and a banner the moment you mark it. New orders also pop up on the Admin page while it's open.

TIP Keep the Admin page open on your phone. Bookmark it, and never share the admin key.
